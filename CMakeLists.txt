cmake_minimum_required(VERSION 3.20)
project(kokoriko VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(KOKORIKO_SANITIZE "Enable address/undefined sanitizers" OFF)
if(NOT JSON_INCLUDE_DIR)
  execute_process(COMMAND "${CMAKE_SOURCE_DIR}/.venv/bin/python" -c "import nlohmann_json; print(nlohmann_json.get_include())" OUTPUT_VARIABLE JSON_INCLUDE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
endif()
add_library(gungi engine/gungi.cpp engine/search.cpp engine/network.cpp)
find_package(Threads REQUIRED)
target_include_directories(gungi PUBLIC engine "${JSON_INCLUDE_DIR}")
target_compile_options(gungi PRIVATE -Wall -Wextra -Wpedantic)
if(KOKORIKO_SANITIZE)
  target_compile_options(gungi PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(gungi PUBLIC -fsanitize=address,undefined)
endif()
add_executable(kokoriko engine/main.cpp)
target_link_libraries(kokoriko PRIVATE gungi Threads::Threads)
add_executable(bench_evaluation EXCLUDE_FROM_ALL scripts/bench_evaluation.cpp)
target_link_libraries(bench_evaluation PRIVATE gungi)
enable_testing()
add_executable(rules_test tests/rules_test.cpp)
target_link_libraries(rules_test PRIVATE gungi)
add_test(NAME rules COMMAND rules_test)
add_executable(search_test tests/search_test.cpp)
target_link_libraries(search_test PRIVATE gungi)
add_test(NAME search COMMAND search_test)
